generated_query.sql Copy
-- only SELECT · validated · run on read-only role
SELECT region,
       SUM(revenue) FILTER (WHERE quarter = 'Q2') AS q2_revenue,
       SUM(revenue) FILTER (WHERE quarter = 'Q3') AS q3_revenue,
       ROUND(100.0 * (q3_revenue - q2_revenue)
             / NULLIF(q2_revenue, 0), 1) AS growth_pct
FROM q3_sales
GROUP BY region
ORDER BY growth_pct DESC;
